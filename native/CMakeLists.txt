cmake_minimum_required(VERSION 3.16)
project(retrocrt_tracking_probe LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The IPC inventory utility is Linux-only. Windows preview needs no native library.")
endif()
add_executable(retrocrt-ipc-inventory src/ipc_inventory.cpp)
target_include_directories(retrocrt-ipc-inventory PRIVATE include)
target_compile_features(retrocrt-ipc-inventory PRIVATE cxx_std_17)
target_compile_options(retrocrt-ipc-inventory PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS retrocrt-ipc-inventory RUNTIME DESTINATION bin)
find_package(Threads REQUIRED)
add_library(retrocrt-ipc-reader STATIC src/ipc_reader.cpp)
target_include_directories(retrocrt-ipc-reader PUBLIC include)
target_compile_features(retrocrt-ipc-reader PUBLIC cxx_std_17)
target_compile_options(retrocrt-ipc-reader PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(retrocrt-ipc-reader PUBLIC Threads::Threads)
add_executable(retrocrt-tracking-bridge src/tracking_bridge.cpp)
target_compile_options(retrocrt-tracking-bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(retrocrt-tracking-bridge PRIVATE retrocrt-ipc-reader)
install(TARGETS retrocrt-tracking-bridge RUNTIME DESTINATION bin)
enable_testing()
add_executable(retrocrt-ipc-reader-test tests/ipc_reader_test.cpp)
target_link_libraries(retrocrt-ipc-reader-test PRIVATE retrocrt-ipc-reader)
target_compile_options(retrocrt-ipc-reader-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME ipc_reader COMMAND retrocrt-ipc-reader-test)
set_tests_properties(ipc_reader PROPERTIES TIMEOUT 10)
