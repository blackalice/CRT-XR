cmake_minimum_required(VERSION 3.16)
project(crt_xr_tracking_probe LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The IPC inventory utility is Linux-only. Windows preview needs no native library.")
endif()
add_executable(crt-xr-ipc-inventory src/ipc_inventory.cpp)
target_include_directories(crt-xr-ipc-inventory PRIVATE include)
target_compile_features(crt-xr-ipc-inventory PRIVATE cxx_std_17)
target_compile_options(crt-xr-ipc-inventory PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS crt-xr-ipc-inventory RUNTIME DESTINATION bin)
find_package(Threads REQUIRED)
add_library(crt-xr-ipc-reader STATIC src/ipc_reader.cpp)
target_include_directories(crt-xr-ipc-reader PUBLIC include)
target_compile_features(crt-xr-ipc-reader PUBLIC cxx_std_17)
target_compile_options(crt-xr-ipc-reader PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(crt-xr-ipc-reader PUBLIC Threads::Threads)
add_executable(crt-xr-tracking-bridge src/tracking_bridge.cpp)
target_compile_options(crt-xr-tracking-bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(crt-xr-tracking-bridge PRIVATE crt-xr-ipc-reader)
install(TARGETS crt-xr-tracking-bridge RUNTIME DESTINATION bin)
enable_testing()
add_executable(crt-xr-ipc-reader-test tests/ipc_reader_test.cpp)
target_link_libraries(crt-xr-ipc-reader-test PRIVATE crt-xr-ipc-reader)
target_compile_options(crt-xr-ipc-reader-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME ipc_reader COMMAND crt-xr-ipc-reader-test)
set_tests_properties(ipc_reader PROPERTIES TIMEOUT 10)
