cmake_minimum_required(VERSION 3.17)
project(retrocrt_libretro LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_USE_STATIC_CPP ON CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/tools/godot-cpp" "${CMAKE_BINARY_DIR}/godot-cpp")
add_library(retrocrt_libretro SHARED retro_core.cpp libretro_host.cpp register_types.cpp)
target_include_directories(retrocrt_libretro PRIVATE "${ROOT}/tools/snes9x/libretro")
target_compile_features(retrocrt_libretro PRIVATE cxx_std_17)
target_compile_options(retrocrt_libretro PRIVATE -Wall -Wextra -Wpedantic -Wno-unused-parameter)
target_link_libraries(retrocrt_libretro PRIVATE godot-cpp ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_options(retrocrt_libretro PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_libraries(retrocrt_libretro PRIVATE winpthread)
    set(OS_NAME windows)
else()
    set(OS_NAME linux)
endif()
set_target_properties(retrocrt_libretro PROPERTIES PREFIX "" OUTPUT_NAME "retrocrt_libretro.${OS_NAME}.x86_64"
    LIBRARY_OUTPUT_DIRECTORY "${ROOT}/extensions/bin" RUNTIME_OUTPUT_DIRECTORY "${ROOT}/extensions/bin")
add_library(retrocrt_mock_core SHARED tests/mock_libretro.cpp)
target_include_directories(retrocrt_mock_core PRIVATE "${ROOT}/tools/snes9x/libretro")
add_executable(retrocrt-host-test tests/host_test.cpp libretro_host.cpp)
target_include_directories(retrocrt-host-test PRIVATE . "${ROOT}/tools/snes9x/libretro")
target_compile_features(retrocrt-host-test PRIVATE cxx_std_17)
target_link_libraries(retrocrt-host-test PRIVATE ${CMAKE_DL_LIBS})
