cmake_minimum_required(VERSION 3.17)
project(crt_xr_libretro LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_USE_STATIC_CPP ON CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/tools/godot-cpp" "${CMAKE_BINARY_DIR}/godot-cpp")
find_package(Threads REQUIRED)
# The video decoder is compiled once per supported FFmpeg major, each against
# that major's headers and in its own namespace; the system FFmpeg is loaded at
# runtime and never linked. video_backend.cpp picks whichever version loads.
set(CRT_XR_FFMPEG_MAJORS 61 62)
set(CRT_XR_VIDEO_OBJECTS)
foreach(major ${CRT_XR_FFMPEG_MAJORS})
    add_library(crt_xr_video_${major} OBJECT video_engine.cpp)
    target_include_directories(crt_xr_video_${major} PRIVATE .)
    target_include_directories(crt_xr_video_${major} SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ffmpeg-${major}/include")
    target_compile_definitions(crt_xr_video_${major} PRIVATE CRT_XR_FFMPEG_NAMESPACE=ffmpeg${major})
    target_compile_features(crt_xr_video_${major} PRIVATE cxx_std_17)
    set_target_properties(crt_xr_video_${major} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(NOT MSVC)
        target_compile_options(crt_xr_video_${major} PRIVATE -Wall -Wextra -Wpedantic -Wno-unused-parameter)
    endif()
    list(APPEND CRT_XR_VIDEO_OBJECTS $<TARGET_OBJECTS:crt_xr_video_${major}>)
endforeach()
add_library(crt_xr_libretro SHARED retro_core.cpp libretro_host.cpp register_types.cpp radio_mp3.cpp video_backend.cpp video_decoder.cpp ${CRT_XR_VIDEO_OBJECTS})
target_include_directories(crt_xr_libretro PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libretro")
target_compile_features(crt_xr_libretro PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(crt_xr_libretro PRIVATE /W4)
    set_property(TARGET crt_xr_libretro PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(crt_xr_libretro PRIVATE -Wall -Wextra -Wpedantic -Wno-unused-parameter)
endif()
target_link_libraries(crt_xr_libretro PRIVATE godot-cpp ${CMAKE_DL_LIBS} Threads::Threads)
if(WIN32)
    if(MINGW)
        target_link_options(crt_xr_libretro PRIVATE -static -static-libgcc -static-libstdc++)
        target_link_libraries(crt_xr_libretro PRIVATE winpthread)
    endif()
    set(OS_NAME windows)
else()
    set(OS_NAME linux)
endif()
set_target_properties(crt_xr_libretro PROPERTIES PREFIX "" OUTPUT_NAME "crt_xr_libretro.${OS_NAME}.x86_64"
    LIBRARY_OUTPUT_DIRECTORY "${ROOT}/extensions/bin" RUNTIME_OUTPUT_DIRECTORY "${ROOT}/extensions/bin"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${ROOT}/extensions/bin")
add_library(crt_xr_mock_core SHARED tests/mock_libretro.cpp)
target_include_directories(crt_xr_mock_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libretro")
add_library(crt_xr_mock_fullpath_core SHARED tests/mock_libretro.cpp)
target_compile_definitions(crt_xr_mock_fullpath_core PRIVATE MOCK_NEED_FULLPATH)
target_include_directories(crt_xr_mock_fullpath_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libretro")
add_executable(crt-xr-host-test tests/host_test.cpp libretro_host.cpp)
target_include_directories(crt-xr-host-test PRIVATE . "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libretro")
target_compile_features(crt-xr-host-test PRIVATE cxx_std_17)
target_link_libraries(crt-xr-host-test PRIVATE ${CMAKE_DL_LIBS})
add_executable(crt-xr-video-test tests/video_test.cpp video_backend.cpp ${CRT_XR_VIDEO_OBJECTS})
target_include_directories(crt-xr-video-test PRIVATE .)
target_compile_features(crt-xr-video-test PRIVATE cxx_std_17)
target_link_libraries(crt-xr-video-test PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
if(MINGW)
    target_link_options(crt-xr-video-test PRIVATE -static -static-libgcc -static-libstdc++)
endif()
